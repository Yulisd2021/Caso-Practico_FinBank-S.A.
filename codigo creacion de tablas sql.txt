
CODIGO CREACION DE TABLAS 

CREATE TABLE TB_CLIENTES_CORE(
id_cli varchar(50) not null PRIMARY KEY,
nomb_cli varchar(50)not null,
apell_cli varchar(50)not null,
tip_doc varchar(50)not null,
num_doc varchar (20)not null,
fec_nac date null,
fec_alta date not null,
cod_segmento varchar(20) not null,
score_buro varchar(50)not null,
ciudad_res varchar(50)not null,
depto_res varchar(50)not null,
estado_cli varchar(50)not null,
canal_adquis varchar(50)not null,
);

CREATE TABLE TB_PRODUCTOS_CAT(

cod_prod varchar(50)not null PRIMARY KEY,
desc_prod varchar(200)not null,
tip_prod varchar(50)not null,
tasa_ea decimal(5,2)not null,
plazo_max_meses int not null,
cuota_min decimal (20,2)not null,
comision_admin decimal (20,2)not null,
estado_prod varchar(50)not null,
);

CREATE TABLE TB_MOV_FINANCIEROS(
id_mov varchar(50)not null PRIMARY KEY,
id_cli varchar(50)not null,
cod_prod varchar (50)not null ,
num_cuenta varchar(50)not null,
fec_mov date not null,
hra_mov time not null,
vr_mov decimal(20,2)not null,
tip_mov varchar(50)not null,
cod_canal varchar(50)not null,
cod_ciudad varchar(50)not null,
cod_estado_mov varchar(50)not null,
id_dispositivo varchar(50)not null, 

);

CREATE TABLE TB_OBLIGACIONES (
id_oblig varchar(50)not null PRIMARY KEY,
id_cli varchar(50)not null,
cod_prod varchar(50)not null,
vr_desembolsado decimal (20,2)not null,
sdo_capital decimal(20,2) not null,
vr_cuota decimal (20,2)not null,
fec_desembolso date not null,
fec_venc date not null,
dias_mora_act int not null,
num_cuotas_pend int not null,
calif_riesgo varchar(50)not null,

);
CREATE TABLE TB_SUCURSALES_RED (
cod_suc varchar(50)not null PRIMARY KEY,
nom_suc varchar(50) not null,
tip_punto varchar(50)not null,
ciudad varchar(50)not null,
depto varchar(50)not null,
latitud decimal (10,5)not null,
longitud decimal (10,5)not null,
activo varchar(50)not null,

);

CREATE TABLE TB_COMISIONES_LOG (
id_comision varchar(50)not null PRIMARY KEY,
id_cli varchar(50)not null,
cod_prod varchar(50)not null,
fec_cobro date not null,
vr_comision decimal(20,2)not null,
tip_comision varchar(50)not null,
estado_cobro varchar(50)not null,
);
